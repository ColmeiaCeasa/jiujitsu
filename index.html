Google Sheets & Sync

Você pode conectar este aplicativo a uma planilha do Google Sheets através de um Web App do Google Apps Script para salvar todas as frequências na nuvem.

Como configurar o Google Sheets:

  1. Crie uma nova planilha no Google Sheets com 3 abas: Alunos, Professores e Frequencia.
  2. Vá em Extensões > Apps Script e cole o script abaixo.
  3. Clique em Implantar > Nova implantação (Tipo: Aplicativo da Web, Acesso: Qualquer pessoa).
  4. Copie a URL gerada e cole no campo acima.
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(data.action);
    if (!sheet) {
      sheet = ss.insertSheet(data.action);
    }
    if (data.action === 'Frequencia') {
      sheet.appendRow([data.date, data.professor, data.studentName, data.category, data.belt, data.status]);
    } else if (data.action === 'Alunos' || data.action === 'Professores') {
      // Clear and rewrite or append
      sheet.appendRow([JSON.stringify(data.payload)]);
    }
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (f) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: f.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
